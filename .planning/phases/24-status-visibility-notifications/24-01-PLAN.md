```typescript
import {Injectable} from "@angular/core";
import {Subject, Observable} from "rxjs";

export interface Toast {
    message: string;
    type: "success" | "info" | "warning" | "danger";
    autohide: boolean;
    delay: number;  // milliseconds
}

/**
 * ToastService manages ephemeral toast notifications.
 * Toasts are emitted via a Subject and consumed by the app component's
 * toast container. Unlike NotificationService (persistent alerts in header),
 * toasts are transient and auto-dismiss.
 */
@Injectable({providedIn: "root"})
export class ToastService {
    private _toasts$ = new Subject<Toast>();

    get toasts$(): Observable<Toast> {
        return this._toasts$.asObservable();
    }

    show(toast: Partial<Toast> & {message: string}): void {
        this._toasts$.next({
            type: toast.type ?? "info",
            autohide: toast.autohide ?? true,
            delay: toast.delay ?? 5000,
            ...toast
        });
    }

    success(message: string): void {
        this.show({message, type: "success"});
    }

    info(message: string): void {
        this.show({message, type: "info"});
    }

    warning(message: string): void {
        this.show({message, type: "warning"});
    }

    danger(message: string): void {
        this.show({message, type: "danger"});
    }
}
```

Design decisions:
- Uses `{providedIn: "root"}` instead of being listed in app.config.ts providers. This is the modern Angular pattern (tree-shakeable) and avoids touching the providers array.
- Uses Subject (not BehaviorSubject) because toasts are ephemeral events, not persistent state. New subscribers don't need to see past toasts.
- The `show()` method accepts `Partial<Toast> & {message: string}` so callers can provide just a message and type, with defaults for autohide and delay.
- Convenience methods (`success()`, `info()`, etc.) mirror Bootstrap's color variants.
- 5000ms default delay matches the Phase 24 requirement ("auto-dismiss after 5 seconds").

**Do NOT:**
- Add the ToastService to app.config.ts providers (providedIn: "root" handles registration)
- Store toast state in the service (toasts are ephemeral, the component manages display lifecycle)

```typescript
        if (index >= 0) {
            this.toasts.splice(index, 1);
        }
    }
```

Design decisions:
- Uses a plain array (not Immutable) because toasts are ephemeral UI state, not shared model data.
- Uses setTimeout for auto-dismiss instead of Bootstrap's toast JavaScript API. This is simpler, avoids DOM querying, and works with Angular's change detection.
- The subscription stores the toast reference so the timeout can remove the exact toast instance.
- AppComponent is the root component and always exists, making it the right place for the global toast container.

- `d-flex` layout with `m-auto` on the close button aligns it vertically centered next to the toast body.
- `aria-live="assertive"` and `role="alert"` ensure screen readers announce toasts.
