export class Config extends ConfigRecord implements IConfig {
    general: IGeneral;
    lftp: ILftp;
    controller: IController;
    web: IWeb;
    autoqueue: IAutoQueue;
    sonarr: ISonarr;

    constructor(props) {
        // Create immutable members
        super({
            general: GeneralRecord(props.general),
            lftp: LftpRecord(props.lftp),
            controller: ControllerRecord(props.controller),
            web: WebRecord(props.web),

    }
}

Add the test connection handler method after `onCommandRestart()`:

```typescript
onTestSonarrConnection(): void {
    this.testConnectionLoading = true;
    this.testConnectionResult = null;

    this._configService.testSonarrConnection().subscribe({
        next: reaction => {
            this.testConnectionLoading = false;
            if (reaction.success) {
                try {
                    const result = JSON.parse(reaction.data);
                    if (result.success) {
                        this.testConnectionResult = {
                            success: true,
                            message: "Connected to Sonarr v" + result.version
                        };
                    } else {
                        this.testConnectionResult = {
                            success: false,
                            message: result.error
                        };
                    }
                } catch {
                    this.testConnectionResult = {
                        success: false,
                        message: "Unexpected response from server"
                    };
                }
            } else {
                this.testConnectionResult = {
                    success: false,
                    message: reaction.errorMessage || "Failed to reach server"
                };
            }
        }
    });
}
```

Design decisions:
- `testConnectionLoading` controls the button text ("Test Connection" vs "Testing...") and disables the button during the request to prevent double-clicks.
- `testConnectionResult` stores the last test result for inline display. Null means no result shown yet. Persists until next test click.
- The handler first checks `reaction.success` (HTTP-level success from RestService). If the HTTP request succeeded, it parses the JSON body to check Sonarr-level success. This two-level check handles both network errors and Sonarr API errors.
- Uses try/catch for JSON.parse as defensive coding against unexpected responses.
- The component uses OnPush change detection. Since the test connection response comes from an Observable subscription, Angular's async pipe and zone.js will trigger change detection. However, the `testConnectionLoading` and `testConnectionResult` are not bound via async pipe — they're direct property bindings. For OnPush, we need change detection to run. The Observable subscription within zone.js should trigger it, but if it doesn't, we may need to inject ChangeDetectorRef and call markForCheck(). Start without it; add if needed during verification.

Do NOT modify any existing methods (onSetConfig, onCommandRestart, ngOnInit).
Do NOT modify the component decorator or existing imports (except adding OptionType to the option.component import).
